precision highp float;
varying vec3 vNormalW;
varying vec2 vUV;
varying float vFaceId;
varying vec3 vPositionW;

uniform vec3 uLightPos;
uniform vec3 uCameraPos;
uniform vec3 uLightColor;

uniform vec4 uFrontFaceColor;
uniform float uFrontAlpha; // uniform float tracking Alpha for THREE.Color

uniform vec4 uSideWallColor;
uniform float uSideAlpha;

uniform vec4 uBackFaceColor;
uniform float uBackAlpha;

uniform vec4 uBackgroundColor;
uniform float uBackgroundAlpha;

uniform vec4 uCornerColor;
uniform float uCornerAlpha;


uniform sampler2D uDiffuseTexture;
uniform sampler2D uDiffuseBackground;
uniform float uHasTexture;
uniform float uHasBackground;

uniform vec3 uEmissiveColor; // Global self-illumination control
uniform vec3 uAmbientColor;
uniform vec3 uDiffuseColor;
uniform vec3 uSpecularColor;
uniform float uSpecularPower;
uniform float uAlpha;
uniform float uDisableLighting;

uniform float uTime;
uniform float uEffectIntensity;
uniform float uEffectSpeed;

void main(void) {
vec4 baseColor = vec4(1.0);
float isBg = 0.0;

// 1. ISOLATE SURFACE CHANNELS VIA INTERPOLATED FACE ID
if (vFaceId < 0.5) {
	baseColor = vec4(uFrontFaceColor, uFrontAlpha);
} else if (vFaceId < 1.5) {
	baseColor = vec4(uSideWallColor, uSideAlpha);
} else if (vFaceId < 2.5) {
	baseColor = vec4(uBackFaceColor, uBackAlpha);
} else if (vFaceId < 3.5) {
	baseColor = vec4(uBackgroundColor, uBackgroundAlpha);
    isBg = 1.0;
} else {
	baseColor = vec4(uCornerColor, uCornerAlpha); // channel mapping for Border/Outline stroke elements
}

// 2. MIX LAYER TEXTURES Symmetrically
if (isBg > 0.5) {
    if (uHasBackground > 0.5) baseColor *= texture2D(uDiffuseBackground, vUV);
} else {
    // Front/Side/Back Typographic bodies and Borders use the text sampler mapping array
    if (uHasTexture > 0.5) baseColor *= texture2D(uDiffuseTexture, vUV);
}

    
    // 3. SEAMLESS LIGHT ROUTING
    vec3 finalRGB = baseColor.rgb;
    
    if (uDisableLighting < 0.5) {
        vec3 lightDir = normalize(uLightPos - vPositionW);
        float diff = max(dot(vNormalW, lightDir), 0.0);
        
        // Correct Additive Lighting Formulation
        vec3 ambientComponent = baseColor.rgb * uAmbientColor;
        vec3 diffuseComponent = baseColor.rgb * uDiffuseColor * diff * uLightColor;
        
        finalRGB = ambientComponent + diffuseComponent;
        
        // Apply Shiny Gloss Highlights selectively to text layers
        if (isBg < 0.5) {
            vec3 viewDir = normalize(uCameraPos - vPositionW);
            vec3 reflectDir = reflect(-lightDir, vNormalW);
            float spec = pow(max(dot(viewDir, reflectDir), 0.0), uSpecularPower);
            finalRGB += uSpecularColor * spec * uLightColor;
        }
    }
    
    // 4. CONSTRAIN THE GLOW LAYER (Only the Bevel Rim glows via uEmissiveColor)
    if (vFaceId >= 0.5 && vFaceId < 1.5) {
        finalRGB += uEmissiveColor; // Injects glow vectors purely into the bevel edges
    }
	
    // THE HARDWARE CLIP: Destroys empty fragments before colors blend
    if (baseColor.a * uAlpha < 0.1) {
        discard; 
    }
    
    // 5. RUNTIME NEON PROCEDURAL EFFECTS
    if (vFaceId < 0.5 && uEffectIntensity > 0.01) {
        float t = uTime * uEffectSpeed;
        float wave = sin(vPositionW.x * 2.0 + t) * cos(vPositionW.y * 2.0 + t);
        vec3 neon = vec3(0.0, 1.0, 0.8);
        finalRGB = mix(finalRGB, neon, wave * uEffectIntensity);
    }
    
    gl_FragColor = vec4(finalRGB, baseColor.a * uAlpha);
}
